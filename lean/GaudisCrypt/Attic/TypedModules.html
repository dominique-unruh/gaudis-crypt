                                          def GaudisCrypt.TypedModules.liftRenaming {Δ Γ : ModuleContext} {A : ModuleTypeRep} (ρ : {T : ModuleTypeRep} → ModuleContextIdx Δ T → ModuleContextIdx Γ T) {T : ModuleTypeRep} :
